Kernel & Package Management

AL2023 uses deterministic updates through versioned repositories for fleet consistency.

Check for Updates

bash 
# List available updated repositories
$ dnf check-release-update

# Check available package updates for a specific release
$ dnf check-update --releasever=2023.0.20230308

Update Packages

bash 
# Full distribution upgrade to a specific version
$ sudo dnf upgrade --releasever=2023.0.20230308

# Update a specific package
$ sudo dnf update <package_name>

# Default behavior: NO updates (locked to AMI version)
$ sudo dnf update
Nothing to do. Complete!

Package Management (DNF)

bash 
# Search for packages
$ dnf search httpd

# Install a package
$ sudo dnf install httpd

# List installed packages
$ dnf list installed

# Remove a package
$ sudo dnf remove httpd

# yum still works (it's a symlink to dnf)
$ which yum
/usr/bin/yum -> dnf-3

bash 
# AMI aliases for AL2023:

Migration

AL2 → AL2023 Key Differences

Critical changes you need to know when migrating from Amazon Linux 2 to AL2023.

Feature Amazon Linux 2 Amazon Linux 2023
Package Manager YUM DNF yum symlinked
Update Model Rolling updates Deterministic (version-locked repos)
Networking ISC dhclient systemd-networkd
Logging rsyslog systemd journal
Cron cron/crond systemd timers
Python Python 2.7 Python 3 only
cgroups v1 v2 (unified hierarchy)
/tmp Disk-backed tmpfs (RAM-backed)
EBS Volume gp2 gp3 (20% cheaper)
Boot Mode Legacy BIOS UEFI Preferred + Secure Boot
Java OpenJDK / Corretto Amazon Corretto (default JVM)
AWS CLI v1 v2
Compiler Triplet aarch64-redhat-linux-gcc aarch64-amazon-linux-gcc
32-bit i686 Limited runtime support Not supported
lsb_release system-lsb-core /etc/os-release only
Desktop MATE GNOME (from 2023.7)
SELinux + cloud-init Not supported together Fully supported
cloud-init errors Silent failure Fatal error (fail closed)
⚠️

Breaking: cloud-init now fails on unavailable remote resources

AL2023 treats unavailable remote resources as fatal errors. Review all cloud-init configurations before migrating.

⚠️

No automatic security updates by default

Unlike AL2, AL2023 does NOT install security updates via cloud-init. Use deterministic upgrading features or manage updates explicitly.

🔴

AL2 End of Life: June 30, 2026

Amazon Linux 2 will stop receiving security patches after this date. Begin migration planning now.

Deploy with IaC Templates

Production-ready Terraform and CloudFormation templates for Amazon Linux 2023.

main.tf 
# ─── Amazon Linux 2023 EC2 Instance ───────────────────────
# Terraform configuration for launching AL2023 with
# dynamic AMI resolution via SSM Parameter Store

terraform {
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }
}

provider "aws" {
  region = var.aws_region
}

variable "aws_region" {
  default = "us-east-2"
}

variable "instance_type" {
  default = "t3.micro"
}

variable "key_name" {
  description = "SSH key pair name"
  type        = string
}

# Dynamic AMI lookup via SSM
data "aws_ssm_parameter" "al2023_ami" {
  name = "/aws/service/ami-amazon-linux-latest/al2023-ami-kernel-default-x86_64"
}

# Security Group
resource "aws_security_group" "al2023_sg" {
  name_prefix = "al2023-"
  
  ingress {
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

# EC2 Instance
resource "aws_instance" "al2023" {
  ami                    = data.aws_ssm_parameter.al2023_ami.value
  instance_type          = var.instance_type
  key_name               = var.key_name
  vpc_security_group_ids = [aws_security_group.al2023_sg.id]

  root_block_device {
    volume_type = "gp3"
    volume_size = 30
    encrypted   = true
  }

  tags = {
    Name = "AL2023-Instance"
    OS   = "Amazon Linux 2023"
  }
}

output "instance_public_ip" {
  value = aws_instance.al2023.public_ip
}

output "ssh_command" {
  value = "ssh ec2-user@${aws_instance.al2023.public_ip}"
}

System Health Check

health-check.sh
bash 
#!/bin/bash
# AL2023 System Health Check Script
set -euo pipefail

echo "═══════════════════════════════════════════"
echo "  Amazon Linux System Health Report"
echo "  $(date '+%Y-%m-%d %H:%M:%S %Z')"
echo "═══════════════════════════════════════════"

# OS Information
echo -e "\n📌 OS Information:"
cat /etc/os-release | grep -E "^(NAME|VERSION|ID)="

# Kernel
echo -e "\n🔧 Kernel:"
uname -r

# Uptime & Load
echo -e "\n⏱️  Uptime & Load:"
uptime

# Memory
echo -e "\n💾 Memory Usage:"
free -h | head -2

# Disk
echo -e "\n💿 Disk Usage:"
df -h / | tail -1

# Security Updates Available
echo -e "\n🔒 Release Update Check:"
dnf check-release-update 2>/dev/null || echo "  System is up to date."

# Network Interfaces
echo -e "\n🌐 Network Interfaces:"
networkctl status 2>/dev/null | head -5 || ip addr show | grep "inet "

# Failed Services
echo -e "\n❌ Failed Services:"
systemctl --failed --no-pager 2>/dev/null || echo "  None"

echo -e "\n═══════════════════════════════════════════"

Safe Update Script

safe-update.sh
bash 
#!/bin/bash
# AL2023 Safe Update Script
# Performs a controlled update with rollback capability
set -euo pipefail

LOGFILE="/var/log/al2023-update-$(date +%Y%m%d).log"

log() { echo "[$(date '+%H:%M:%S')] $*" | tee -a "$LOGFILE"; }

log "Starting AL2023 safe update process..."

# Step 1: Check available updates
log "Checking for available release updates..."
UPDATES=$(dnf check-release-update 2>&1) || true
echo "$UPDATES" | tee -a "$LOGFILE"

if echo "$UPDATES" | grep -q "No release update"; then
  log "✅ System is already on the latest release."
  exit 0
fi

# Step 2: Extract latest version
LATEST=$(echo "$UPDATES" | grep "Version " | tail -1 | awk '{print $2}' | tr -d ':')
log "Latest available version: $LATEST"

# Step 3: Dry run
log "Performing dry-run upgrade to $LATEST..."
dnf upgrade --releasever="$LATEST" --assumeno 2>&1 | tee -a "$LOGFILE" || true

# Step 4: Prompt for confirmation
read -p "Proceed with upgrade to $LATEST? (y/N): " CONFIRM
if [[ "$CONFIRM" != "y" && "$CONFIRM" != "Y" ]]; then
  log "❌ Update cancelled by user."
  exit 1
fi

# Step 5: Create snapshot marker
log "Recording pre-update state..."
rpm -qa --queryformat '%{NAME}-%{VERSION}-%{RELEASE}.%{ARCH}\n' \
  | sort > "/tmp/packages-before-update.txt"

# Step 6: Execute upgrade
log "Upgrading to $LATEST..."
dnf upgrade --releasever="$LATEST" -y 2>&1 | tee -a "$LOGFILE"

log "✅ Update complete. Reboot recommended."
log "Review log at: $LOGFILE"

Container Setup

container-setup.sh
bash 
#!/bin/bash
# Setup Docker with AL2023 on an EC2 instance
set -euo pipefail

echo "🐳 Installing Docker on Amazon Linux 2023..."

# Install Docker
sudo dnf install -y docker
sudo systemctl enable --now docker
sudo usermod -aG docker ec2-user

# Login to ECR Public
aws ecr-public get-login-password --region us-east-1 \
  | docker login --username AWS --password-stdin public.ecr.aws

# Pull Amazon Linux images
echo "📦 Pulling Amazon Linux container images..."
docker pull public.ecr.aws/amazonlinux/amazonlinux:2023
docker pull public.ecr.aws/amazonlinux/amazonlinux:2027-minimal

# Verify
echo "✅ Images available:"
docker images | grep amazonlinux

echo ""
echo "🎉 Docker setup complete!"
echo "Run: docker run --rm -it amazonlinux:2023 /bin/bash"

Networking

EC2 Network Utilities

amazon-ec2-net-utils v2.x — systemd-networkd based network management for AL2023+

What it does

  • Per-interface policy routing for VPC source/dest restrictions
  • Configuration of secondary IPv4 addresses
  • ENI hotplug configuration
  • Routing for delegated prefixes
  • Generated configs stored in /run/ (ephemeral)

Verification Commands

# Verify networkd is running
$ networkctl
# Should show interfaces as "routable" and "configured"

# Check interface config source
$ networkctl status ens5
# Look for: /run/systemd/network/70-ens5.network

# Check DNS resolution
$ resolvectl
